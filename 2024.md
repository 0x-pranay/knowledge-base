- rootTagId is used to get/update/delete the rootTag and all its tags.
- for every 1 rootTag there will be N tags. where N= no of entities associated with the rootTag's attribute.

tags

- tags are created internally when rootTags are created or when a new entity is added to an attribute.

- `tagId` are actually used to attach a tag to an actual trip/driver/asset/event

  if (doc.docType === 'trip') {
    emit([doc.tripId], { ongoing: doc.ongoing });

      totalCount: 1,
      uploadedCount: isWebhookTriggered ? 1 : 0,
      nonUploadedWebhookTriggerCount: nonUploadedWebhookTriggerEligible ? 1 : 0,
    });
  } else if (doc.docType === 'route') {
    var NON_UPLOADED_WEBHOOK_TRIGGER_STATUSES = ['NOT_MARKED_FOR_UPLOAD', 'CANCELLED'];

    });

}

/** Returns true if the input status needs to be considered for the trip webhook */
function isFileEligibleForWebhookTrigger(files, status) {
  if (!(files && files.length)) {
    return false;
  }

  var mediaFilesEligibleForWebhookTrigger = [];
  for (var index = 0; index < files.length; index++) {
    var file = files[index];
    if (file.uploadStats) {
      file.uploadStats.status === status && mediaFilesEligibleForWebhookTrigger.push(file);
    }
  }

  return files.length === mediaFilesEligibleForWebhookTrigger.length;
}

CREATE OR REPLACE FUNCTION staging.process_staging(
	p_clientid character varying,
	p_database character varying,
	p_batch integer,
	p_jsonb jsonb,
	p_seq_no text)
    RETURNS boolean
    LANGUAGE 'plpgsql'
    COST 100
    VOLATILE PARALLEL UNSAFE
AS $BODY$
DECLARE
v_processtime integer;
v_tablename VARCHAR(1000);
v_return BOOLEAN;
v_temp TEXT; 
SPACE CONSTANT CHAR(1) =TRIM('');
vCustomerDebug BOOLEAN;
v_returnjsonb JSONB;
v_warehouseenabled BOOLEAN;
v_starttime timestamp := (clock_timestamp() at time zone 'utc');
v_failed BOOLEAN;
BEGIN

    
    p_clientid := TRIM (p_clientid);
    p_database := TRIM (p_database);
	
	p_jsonb :=p_jsonb-'_rev';
	
	/*If batch is NULL, assume it is coming from replicator */
	IF p_batch IS NULL THEN
		p_batch =-1;
	END IF;

    CALL EXTENSIONS.SET_SEARCH_PATHX(p_clientid);
	v_warehouseenabled := TRUE;
--     SELECT WAREHOUSE_ENABLED FROM STAGING.CLIENT WHERE LOWER(CLIENTID)=  LOWER(p_clientid) INTO v_warehouseenabled;

	
    /*If there is a no row for the client and database , insert the row */ 
   IF ((COALESCE(p_seq_no,SPACE) != SPACE) AND (p_batch !=1))  THEN
  			
		SELECT SEQ FROM STAGING.SEQ WHERE LOWER(CLIENTID)=  LOWER(p_clientid) AND LOWER(CLOUDANT_DATABASE)= LOWER(p_database) INTO v_temp;
		IF COALESCE(v_temp,SPACE) =SPACE THEN
				INSERT INTO STAGING.SEQ(CLIENTID,CLOUDANT_DATABASE) VALUES (p_clientid,p_database) ON CONFLICT (CLIENTID,CLOUDANT_DATABASE) DO NOTHING;
		END IF;
	END IF;

    /*If it is a debug document, determine whether it is important for the customer. These events are stored in the event table itself */
    vCustomerDebug:=FALSE; 
    IF   TRIM(p_jsonb->>'docType') = TRIM('debug') THEN 
-- 	  SELECT EXISTS( SELECT 1   FROM FLEETDRIVER.EVENT_MASTER WHERE TYPE=2 AND TRIM(LOWER(VIOLATION)) = LOWER(p_jsonb->>'eventType') ) INTO vCustomerDebug;
		SELECT EXISTS (SELECT 1 FROM STAGING.CLIENT WHERE CLIENTID = p_clientid AND p_jsonb->>'eventType' = ANY(ALLOWED_DEBUG_EVENTS) ) INTO vCustomerDebug;
    END IF;
	
	/*For failed trips, only save if postgres failed flag is TRUE */
	IF LEFT(p_database,12) =TRIM('failed_trips') THEN 
		    v_failed := COALESCE(CONVERT_BOOLEAN(p_jsonb->>'postgresFlag'),FALSE);
			IF v_failed=FALSE THEN
				RETURN TRUE;
			END IF;
	END IF;
	
    /*Save the design documents in a separate table */
    IF LEFT(p_jsonb->>'_id', 8) = '_design/' THEN
            
            IF p_jsonb->>'_id' IS NOT NULL THEN 
                DELETE FROM STAGING.DESIGN WHERE ID= p_jsonb->>'_id' AND LOWER(CLIENTID)=  LOWER(p_clientid) AND LOWER(CLOUDANT_DATABASE)= LOWER(p_database);
                INSERT INTO STAGING.DESIGN (ID,CLIENTID,CLOUDANT_DATABASE,PAYLOAD) VALUES (p_jsonb->>'_id',p_clientid,p_database,p_jsonb) ;
            END IF;
            v_return =  TRUE;
     ELSIF p_database ='camfirmware' THEN
            
            IF p_jsonb->>'_id' IS NOT NULL THEN 
                DELETE FROM STAGING.CAMFIRMWARE WHERE ID= p_jsonb->>'_id' AND LOWER(CLIENTID)=  LOWER(p_clientid) AND LOWER(CLOUDANT_DATABASE)= LOWER(p_database);
                INSERT INTO STAGING.CAMFIRMWARE (ID,CLIENTID,CLOUDANT_DATABASE,PAYLOAD) VALUES (p_jsonb->>'_id',p_clientid,p_database,p_jsonb) ;
            END IF;
            v_return =  TRUE;
			v_tablename ="CAMFIRMWARE";
    ELSIF CONVERT_BOOLEAN( p_jsonb->>'_deleted') = True THEN
             v_return =  TRUE;
    ELSIF (p_jsonb->>'docType') = 'trip' THEN 
                    /*This is a trip document */
        SELECT  STAGING.PROCESS_TRIP(p_clientid,v_warehouseenabled,p_batch,p_jsonb)  INTO v_return;
        v_tablename='TRIP'  ;

    ELSIF ((p_jsonb->>'docType') = 'violation' OR (p_jsonb->>'docType') = 'external' )  THEN 
                   /*This is a event document */ 
            SELECT  STAGING.PROCESS_EVENT(p_clientid,v_warehouseenabled,p_batch,p_jsonb)  INTO v_return; 
		v_tablename='EVENT'; 
    ELSIF (p_jsonb->>'docType') = 'debug'    THEN 
                   /*This is a debug customer document */
            SELECT STAGING.PROCESS_DEBUG_CUSTOMER(p_clientid,v_warehouseenabled,p_batch,p_jsonb)  INTO v_return;
    	v_tablename='DEBUG_CUSTOMER_EVENT';
--  	ELSIF (p_jsonb->>'docType') = 'debug'  AND  vCustomerDebug= False  THEN 
--                    /*This is a debug customer document */
--             SELECT STAGING.PROCESS_DEBUG(p_clientid,p_batch,p_jsonb)  INTO v_return;
--     	v_tablename='DEBUG_EVENT';
	 ELSIF (p_jsonb->>'docType') = 'route'    THEN 
                   /*This is a debug customer document */
            SELECT STAGING.PROCESS_ROUTE(p_clientid,False,p_batch,p_jsonb)  INTO v_return;
    	v_tablename='ROUTE';
    ELSIF (SUBSTRING(p_database,1,6) =TRIM('assets')) AND (p_jsonb->>'docType' ='fleetInfo')  THEN  
            SELECT  FLEETDRIVER.F_UPSERT_FLEET(p_clientid,p_batch,p_jsonb)  INTO v_returnjsonb;
            IF CONVERT_INTEGER(v_returnjsonb->>'code')= 201 THEN
                    v_return :=TRUE;
            END IF;
		v_tablename='FLEET';
    ELSIF (SUBSTRING(p_database,1,6) =TRIM('assets')) AND (p_jsonb->>'docType' != 'fleetInfo') THEN  
            /* Asset database processing */
            SELECT  STAGING.PROCESS_ASSET(p_clientid,p_batch,p_jsonb)  INTO v_return;
		v_tablename='ASSET';
    ELSIF (SUBSTRING(p_database,1,3) =TRIM('fr_')) THEN
            /* Face Recognition database processing */
            SELECT  STAGING.PROCESS_FR(p_clientid,p_batch,p_jsonb)  INTO v_return;
    	v_tablename='FR';
	ELSIF  p_jsonb->>'docType' = 'uploadRequest' THEN  
	
		SELECT STAGING.process_dvr(p_clientid,p_batch,p_jsonb)  INTO v_return;
    		v_tablename='STAGING.DVR_EDVR';
		
		v_tablename='DVR_EDVR';
		v_return =  TRUE;
	ELSIF  p_jsonb->>'docType' = 'deviceHealth' THEN  
	
		--INSERT INTO DEVICE_HEALTH_HISTORY(CLIENTID,ingestiontimestamp,PAYLOAD,FLEETID,ASSETID,DEVICEID,EVENTTYPE,MESSAGETYPE)

CREATE OR REPLACE FUNCTION staging.process_debug_customer(
	p_clientid character varying,
	p_warehouseenabled boolean,
	p_batch integer,
	p_jsonb jsonb)
    RETURNS boolean
    LANGUAGE 'plpgsql'
    COST 100
    VOLATILE PARALLEL UNSAFE
AS $BODY$
DECLARE
v_return BOOLEAN;  
 SPACE CONSTANT CHAR(1) := TRIM('');
v_id VARCHAR(3000);
v_debugid VARCHAR(3000);
v_doccreatedat TIMESTAMP;
v_docupdatedat TIMESTAMP;
v_doccreatedat_db TIMESTAMP;
v_docupdatedat_db TIMESTAMP;
v_returnjsonb JSONB;
BEGIN
    CALL EXTENSIONS.SET_SEARCH_PATHX(p_clientid);
	
	IF NOT EXISTS (SELECT 1 FROM STAGING.CLIENT WHERE CLIENTID=p_clientid AND  p_jsonb->>'eventType' = ANY(ALLOWED_DEBUG_EVENTS)) 
	THEN
		RETURN TRUE;
	END IF;
	
	IF p_jsonb->>'tripId' IS NOT NULL THEN
			-- here sdk is needed for further operations so review it once
			p_jsonb =p_jsonb - 'asset' -'sdk'- 'device';
	END IF;
	
    SELECT ID,DOCCREATEDAT,DOCUPDATEDAT 
	FROM S_DEBUG_EVENT_CUSTOMER 
	INTO v_debugid,v_doccreatedat_db,v_docupdatedat_db
    WHERE ID= p_jsonb->>'_id' AND CLIENTID= p_clientid;
    
    v_doccreatedat:=  COALESCE(CONVERT_TIMESTAMP(p_jsonb->>'docCreatedAt')) ;
    v_docupdatedat:=   COALESCE(CONVERT_TIMESTAMP(p_jsonb->>'docUpdatedAt'),v_docupdatedat_db,v_doccreatedat,v_doccreatedat_db) ;

                        /*Event already exists . If incoming document is later, then replace our copy with incoming document*/
	   IF v_docupdatedat_db < CONVERT_TIMESTAMP(p_jsonb->>'docUpdatedAt') THEN 
            UPDATE S_DEBUG_EVENT_CUSTOMER SET PAYLOAD=  p_jsonb, DOCUPDATEDAT= v_docupdatedat,MODIFIEDDATE= (now() at time zone 'utc'), BATCH= p_batch
            WHERE ID= p_jsonb->>'_id'  AND  CLIENTID=p_clientid RETURNING ID INTO v_id; 
       ELSE
            RETURN TRUE;
       END IF;

    ELSE
             INSERT INTO S_DEBUG_EVENT_CUSTOMER
                (ID, CLIENTID, PAYLOAD, DOCCREATEDAT, DOCUPDATEDAT,  CREATEDATE, MODIFIEDDATE, BATCH, TRIPID, FLEETID)

                p_jsonb->>'_id',  p_clientid, p_jsonb, v_doccreatedat, v_docupdatedat,(now() at time zone 'utc'), (now() at time zone 'utc'),p_batch,p_jsonb->>'tripId',p_jsonb->>'fleetId' RETURNING ID INTO v_id;
                
    END IF; 
--END IF;           
    RETURN FALSE;

END ;
